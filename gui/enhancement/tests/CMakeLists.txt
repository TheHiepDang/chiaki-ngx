cmake_minimum_required(VERSION 3.16)
project(chiaki_enhancement_tests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Test Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PLACEBO REQUIRED IMPORTED_TARGET libplacebo>=6.338)
add_executable(enhancement-bridge-test bridge.cpp ../pipeline.cpp ../pipeline.h)
target_compile_features(enhancement-bridge-test PRIVATE cxx_std_11)
target_link_libraries(enhancement-bridge-test PRIVATE Qt6::Core Qt6::Gui PkgConfig::PLACEBO)
add_executable(enhancement-qml-test qml.cpp)
target_compile_features(enhancement-qml-test PRIVATE cxx_std_17)
target_compile_definitions(enhancement-qml-test PRIVATE ENHANCEMENT_QML="${CMAKE_CURRENT_SOURCE_DIR}/../../src/qml/EnhancementSettings.qml")
target_link_libraries(enhancement-qml-test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
add_executable(eye-candy-tone-test tone.cpp)
target_compile_features(eye-candy-tone-test PRIVATE cxx_std_17)
target_link_libraries(eye-candy-tone-test PRIVATE Qt6::Core Qt6::Gui PkgConfig::PLACEBO)
add_executable(enhancement-model-test model.cpp)
target_compile_features(enhancement-model-test PRIVATE cxx_std_11)
enable_testing()
add_test(NAME enhancement-model COMMAND enhancement-model-test)
add_test(NAME enhancement-qml COMMAND enhancement-qml-test)
set_tests_properties(enhancement-qml PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")

add_executable(enhancement-diagnostics-test diagnostics.cpp ../diagnostics.cpp ../diagnostics.h)
target_compile_features(enhancement-diagnostics-test PRIVATE cxx_std_17)
target_link_libraries(enhancement-diagnostics-test PRIVATE Qt6::Core)
add_test(NAME enhancement-diagnostics COMMAND enhancement-diagnostics-test)

add_executable(enhancement-runtime-manifest-test runtime_manifest.cpp ../runtime_manifest.h)
target_compile_features(enhancement-runtime-manifest-test PRIVATE cxx_std_11)
target_link_libraries(enhancement-runtime-manifest-test PRIVATE Qt6::Core)
add_test(NAME enhancement-runtime-manifest COMMAND enhancement-runtime-manifest-test)

add_executable(update-checker-test updates.cpp ../../src/updatechecker.cpp ../../include/updatechecker.h ../diagnostics.cpp)
target_compile_features(update-checker-test PRIVATE cxx_std_17)
target_include_directories(update-checker-test PRIVATE ../../include)
target_link_libraries(update-checker-test PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test)
add_test(NAME update-checker COMMAND update-checker-test)

add_executable(update-ui-test update_ui.cpp)
target_compile_features(update-ui-test PRIVATE cxx_std_17)
target_compile_definitions(update-ui-test PRIVATE UPDATE_QML_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../../src/qml/UpdateBar.qml")
target_link_libraries(update-ui-test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
add_test(NAME update-ui COMMAND update-ui-test)
set_tests_properties(update-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QUICK_CONTROLS_STYLE=Material;QT_FORCE_STDERR_LOGGING=1")

# Fail stalled focused checks explicitly instead of exhausting a release build job.
set_tests_properties(enhancement-diagnostics update-checker PROPERTIES TIMEOUT 120)
