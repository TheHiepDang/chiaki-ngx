cmake_minimum_required(VERSION 3.16)
project(chiaki_enhancement_worker LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The enhancement worker requires Windows x64")
endif()
if(NOT MSVC)
    message(FATAL_ERROR "NGX parameter objects require the Windows MSVC C++ ABI. Build the worker with MSVC, even when the GUI uses MinGW.")
endif()
set(CHIAKI_NGX_SDK "" CACHE PATH "NVIDIA DLSS SDK root (include/nvsdk_ngx.h)")
if(NOT EXISTS "${CHIAKI_NGX_SDK}/include/nvsdk_ngx.h")
    message(FATAL_ERROR "Set CHIAKI_NGX_SDK to a separately obtained NVIDIA DLSS SDK")
endif()

file(READ "third-party/fsr1/ffx_a.h" FSR_A)
file(READ "third-party/fsr1/ffx_fsr1.h" FSR_ONE)
configure_file(worker_shaders.h.in worker_shaders.expanded @ONLY)
# MSVC limits individual string literals. Concatenate bounded chunks at startup.
file(READ "${CMAKE_CURRENT_BINARY_DIR}/worker_shaders.expanded" SHADER_SOURCE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/worker_shaders.h" "#pragma once\n#include <string>\n")
foreach(NAME commonShader fsrShader)
    set(MARKER "static const char *${NAME} = R\"CNGX(")
    string(FIND "${SHADER_SOURCE}" "${MARKER}" BEGIN)
    string(LENGTH "${MARKER}" MARKER_LENGTH)
    math(EXPR BEGIN "${BEGIN} + ${MARKER_LENGTH}")
    string(SUBSTRING "${SHADER_SOURCE}" ${BEGIN} -1 CONTENT)
    string(FIND "${CONTENT}" ")CNGX\";" END)
    string(SUBSTRING "${CONTENT}" 0 ${END} CONTENT)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/worker_shaders.h" "static const std::string ${NAME}Storage = std::string()")
    string(LENGTH "${CONTENT}" LENGTH)
    while(LENGTH GREATER 0)
        if(LENGTH GREATER 12000)
            set(COUNT 12000)
        else()
            set(COUNT ${LENGTH})
        endif()
        string(SUBSTRING "${CONTENT}" 0 ${COUNT} CHUNK)
        file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/worker_shaders.h" " + R\"CNGX(${CHUNK})CNGX\"")
        string(SUBSTRING "${CONTENT}" ${COUNT} -1 CONTENT)
        string(LENGTH "${CONTENT}" LENGTH)
    endwhile()
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/worker_shaders.h" ";\nstatic const char *${NAME} = ${NAME}Storage.c_str();\n")
endforeach()
add_executable(chiaki-enhancement-worker worker.cpp)
target_compile_features(chiaki-enhancement-worker PRIVATE cxx_std_17)
target_include_directories(chiaki-enhancement-worker PRIVATE
    "${CHIAKI_NGX_SDK}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(chiaki-enhancement-worker PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(chiaki-enhancement-worker PRIVATE d3d12 dxgi d3dcompiler advapi32)
set_target_properties(chiaki-enhancement-worker PROPERTIES OUTPUT_NAME nvngx SUFFIX ".dll")
if(MSVC)
    target_compile_options(chiaki-enhancement-worker PRIVATE /W4 /EHsc /utf-8)
endif()
install(TARGETS chiaki-enhancement-worker RUNTIME DESTINATION bin/enhancement)

include(CTest)
add_executable(enhancement-model-test tests/model.cpp)
target_compile_features(enhancement-model-test PRIVATE cxx_std_11)
add_test(NAME enhancement-model COMMAND enhancement-model-test)
